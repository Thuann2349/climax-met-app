  <!-- Lógica da Aplicação -->
  <script>
    let lat = -23.5505, lon = -46.6333, totalConsultas = 0;
    const map = L.map('map').setView([lat, lon], 8);

    L.tileLayer('https://{s}.basemaps.cartocdn.com/dark_all/{z}/{x}/{y}{r}.png', { maxZoom: 18 }).addTo(map);
    L.tileLayer('https://tilecache.rainviewer.com/v2/radar/nowcast_latest/256/{z}/{x}/{y}/2/1_1.png', { opacity: 0.7 }).addTo(map);
    let marker = L.marker([lat, lon]).addTo(map);

    map.on('click', e => atualizarLocalizacao(e.latlng.lat, e.latlng.lng, "Coordenada Geográfica"));

    async function buscarLocalizacao() {
      const city = document.getElementById('city-input').value;
      if (!city) return;
      try {
        const res = await fetch(`https://geocoding-api.open-meteo.com/v1/search?name=${encodeURIComponent(city)}&count=1&language=pt&format=json`);
        const data = await res.json();
        if (data.results && data.results.length > 0) {
          atualizarLocalizacao(data.results[0].latitude, data.results[0].longitude, data.results[0].name);
        } else { alert('Localidade não encontrada.'); }
      } catch (e) { console.error(e); }
    }

    async function atualizarLocalizacao(newLat, newLon, nome) {
      lat = newLat; lon = newLon;
      marker.setLatLng([lat, lon]);
      map.panTo([lat, lon]);
      document.getElementById('coords-display').innerText = `Lat: ${lat.toFixed(4)} | Lon: ${lon.toFixed(4)}`;
      totalConsultas++;
      document.getElementById('admin-q-count').innerText = totalConsultas;
      await carregarClima(nome);
    }

    async function carregarClima(nomeLocal) {
      try {
        const url = `https://api.open-meteo.com/v1/forecast?latitude=${lat}&longitude=${lon}&current=temperature_2m,relative_humidity_2m,surface_pressure,wind_speed_10m,wind_gusts_10m,precipitation&hourly=precipitation_probability&forecast_days=1&timezone=auto`;
        const res = await fetch(url);
        const data = await res.json();

        const cur = data.current;
        document.getElementById('temp-val').innerText = `${cur.temperature_2m} °C`;
        document.getElementById('wind-val').innerText = `${cur.wind_speed_10m} (raj. ${cur.wind_gusts_10m}) km/h`;
        document.getElementById('press-val').innerText = `${cur.surface_pressure} hPa`;
        
        const prob = data.hourly.precipitation_probability ? data.hourly.precipitation_probability[0] : 0;
        document.getElementById('precip-val').innerText = `${prob} %`;

        // Motor de Alarme Preditivo Inteligente
        const card = document.getElementById('alert-card');
        const badge = document.getElementById('risk-badge');
        const msg = document.getElementById('alert-msg');

        if (cur.wind_gusts_10m > 45 || prob > 70) {
          card.className = "glass-panel border-l-4 border-red-500 rounded-xl p-4 transition-all glow-amber";
          badge.className = "text-xs font-semibold px-2 py-0.5 rounded bg-red-950 text-red-400 border border-red-800 animate-pulse";
          badge.innerText = "ALERTA SEVERO DE TEMPESTADE";
          msg.innerText = `Atenção em ${nomeLocal}! Detectada frente de rajadas de ${cur.wind_gusts_10m} km/h e alta instabilidade convectiva. Alarme disparado.`;
          
          // Toca alarme sonoro
          const audio = document.getElementById('alarm-sound');
          if(audio) audio.play().catch(e => console.log("Áudio bloqueado pelo navegador até interação"));
        } else {
          card.className = "glass-panel border-l-4 border-emerald-500 rounded-xl p-4 transition-all";
          badge.className = "text-xs font-semibold px-2 py-0.5 rounded bg-emerald-950 text-emerald-400 border border-emerald-800";
          badge.innerText = "Estabilidade Normal";
          msg.innerText = `Condições normais em ${nomeLocal}. Sem riscos imediatos nas próximas horas.`;
        }

        // Relatório Executivo
        document.getElementById('ai-report').innerHTML = `
          <strong>[RELATÓRIO DE INTELIGÊNCIA AETHERIS]</strong><br>
          - Local: ${nomeLocal} (${lat.toFixed(2)}, ${lon.toFixed(2)})<br>
          - Temperatura Atual: ${cur.temperature_2m}°C<br>
          - Pressão Atmosférica: ${cur.surface_pressure} hPa<br>
          - Probabilidade de Precipitação: ${prob}%<br>
          - Diagnóstico: Atmosfera monitorada em tempo real com matriz preditiva de alta precisão. Nenhuma anomalia crítica oculta nos radares doppler.
        `;
      } catch (err) {
        console.error(err);
      }
    }

    function baixarRelatorio() {
      const texto = document.getElementById('ai-report').innerText;
      const blob = new Blob([texto], { type: 'text/plain;charset=utf-8' });
      const link = document.createElement('a');
      link.href = URL.createObjectURL(blob);
      link.download = 'Relatorio_Meteorologico_Aetheris.txt';
      link.click();
    }
